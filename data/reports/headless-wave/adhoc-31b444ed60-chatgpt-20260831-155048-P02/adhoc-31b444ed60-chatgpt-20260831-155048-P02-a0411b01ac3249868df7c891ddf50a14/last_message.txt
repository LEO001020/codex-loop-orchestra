$ErrorActionPreference = "Stop"

# P02 setup.ps1 — Windows first-local-setup for cloned repo
# Goal: env check + cloudflared download + fixed-pnpm install + Codex Skill + quick tunnel + c2c setup --json + parse JSON output
# No side-effects, no UI, no login, no UAC, no interactive prompts

function Get-Cloudflare-Cloudflared {
    param([string]$Version = "latest")
    $url = "https://github.com/cloudflare/cloudflare-one/releases/latest/download/cloudflared-windows-amd64.exe"
    $exe = "cloudflared-windows-amd64.exe"
    if (-not (Test-Path $exe)) {
        Invoke-WebRequest -Uri $url -OutFile $exe -UseBasicParsing
    }
    return $exe
}

function Install-Pnpm-Fixed {
    $raw = Get-Content -Path "package.json" -Raw | ConvertFrom-Json
    $pnpmVer = $raw.packageManager -replace '^pnpm@', ''
    if (-not $pnpmVer) { throw "package.json missing packageManager" }
    $pnpmPath = "pnpm$pnpmVer"
    if (-not (Test-Path $pnpmPath)) {
        $npm = Join-Path (Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*" | Where-Object { $_.DisplayName -like "Node.js*" } | Select-Object -First 1).InstallLocation "npm.cmd"
        if (-not $npm) { throw "Node.js not found" }
        & $npm install -g pnpm@$pnpmVer
    }
    return $pnpmPath
}

function Install-Codex-Skill {
    $dest = "skill"
    if (-not (Test-Path $dest)) { New-Item -ItemType Directory -Path $dest -Force | Out-Null }
    $files = @("SKILL.md", "plugin.json")
    foreach ($f in $files) {
        $src = Join-Path ".." $f
        if (Test-Path $src) {
            Copy-Item $src $dest -Force
        }
    }
}

$cloudflared = Get-Cloudflare-Cloudflared
$pnpm = Install-Pnpm-Fixed

& ".\bin\c2c.js" tunnel choose --mode quick --json | Out-Null

# c2c setup --json
$json = & ".\bin\c2c.js" setup --json

Write-Output $json
